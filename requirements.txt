# WebInject2026 -- pinned for reproducibility.
#
# These are exact versions, not lower bounds: commit history (see the v4.5
# regeneration) shows that lbfgs converges to marginally different coefficients
# under different numpy/scipy/BLAS builds, which moves threshold-sensitive
# counts by a row and blows up small-denominator ratios. A lower bound (>=)
# permits exactly that drift; the corpus fingerprint identifies the corpus a
# result was scored against but not the stack it ran on, so the stack is pinned
# here instead. Every shipped number was produced against this set.
#
# Python 3.12 is the floor. Two independent reasons: 24 rows carry raw NUL bytes
# from the null-byte evasion and 3.10's csv module rejects NUL on read and write;
# and the pinned scientific stack below (scipy 1.18) requires >=3.12, so 3.11
# cannot install the versions the published numbers were produced against.
# Developed on CPython 3.14.5; CI covers 3.12 and 3.13.
numpy==2.4.6

joblib==1.5.3            # scikit-learn parallelism; pinned for determinism

# Parquet export (data/webinject2026.parquet), written by `python -m src.build`
pyarrow==25.0.0
# Only for `python -m src.build --capture` and `python -m src.crs_baseline`
# (live HTTP replay against a container).
